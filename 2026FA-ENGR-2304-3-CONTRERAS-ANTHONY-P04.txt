#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    int Vs, R1, R2, R3;
    int choice = 1;

    double D;
    double i1, i2;
    double V1, V2, V3;

    double user_i1, user_i2;
    double user_V1, user_V2, user_V3;

    srand(time(0));

    while (choice != 0)
    {
        /* gen circuit values */
        Vs = rand() % 10 + 1;
        R1 = rand() % 20 + 1;
        R2 = rand() % 20 + 1;
        R3 = rand() % 20 + 1;

        /* solve*/
        D = ((R1 + R2) * (R2 + R3)) - (R2 * R2);

        i1 = (Vs * (R2 + R3)) / D;
        i2 = (Vs * R2) / D;

        V1 = R1 * i1;
        V2 = R2 * (i1 - i2);
        V3 = R3 * i2;

        /* truncate */
        i1 = ((int)(i1 * 1000)) / 1000.0;
        i2 = ((int)(i2 * 1000)) / 1000.0;

        V1 = ((int)(V1 * 1000)) / 1000.0;
        V2 = ((int)(V2 * 1000)) / 1000.0;
        V3 = ((int)(V3 * 1000)) / 1000.0;

        printf("\n--- Electrical Circuit Problem ---\n");

        printf("\nVs = %d V\n", Vs);
        printf("R1 = %d Ohms\n", R1);
        printf("R2 = %d Ohms\n", R2);
        printf("R3 = %d Ohms\n", R3);

        printf("\nEnter i1: ");
        scanf("%lf", &user_i1);

        printf("Enter i2: ");
        scanf("%lf", &user_i2);

        printf("Enter V1: ");
        scanf("%lf", &user_V1);

        printf("Enter V2: ");
        scanf("%lf", &user_V2);

        printf("Enter V3: ");
        scanf("%lf", &user_V3);

        /* truncate answers */
        user_i1 = ((int)(user_i1 * 1000)) / 1000.0;
        user_i2 = ((int)(user_i2 * 1000)) / 1000.0;

        user_V1 = ((int)(user_V1 * 1000)) / 1000.0;
        user_V2 = ((int)(user_V2 * 1000)) / 1000.0;
        user_V3 = ((int)(user_V3 * 1000)) / 1000.0;

        printf("\n--- Results ---\n");

        if (user_i1 == i1)
            printf("i1: Correct\n");
        else
            printf("i1: Incorrect. Answer = %.3f\n", i1);

        if (user_i2 == i2)
            printf("i2: Correct\n");
        else
            printf("i2: Incorrect. Answer = %.3f\n", i2);

        if (user_V1 == V1)
            printf("V1: Correct\n");
        else
            printf("V1: Incorrect. Answer = %.3f\n", V1);

        if (user_V2 == V2)
            printf("V2: Correct\n");
        else
            printf("V2: Incorrect. Answer = %.3f\n", V2);

        if (user_V3 == V3)
            printf("V3: Correct\n");
        else
            printf("V3: Incorrect. Answer = %.3f\n", V3);

        printf("\nEnter 1 for another problem or 0 to exit: ");
        scanf("%d", &choice);
    }

    printf("\nProgram closed.\n");

    return 0;
}
